🚀 Complete Bug Bounty Workflow (Advanced)

1. Advanced Subdomain Enum

Comprehensive subdomain enumeration
$ subfinder -d {domain} -all -cs -recursive | tee subfinder.txt && amass enum -d {domain} -active -brute -w /path/to/wordlist.txt -rf /path/to/resolvers.txt -timeout 60 | tee amass.txt && findomain -t {domain} --quiet | tee findomain.txt && assetfinder --subs-only {domain} | tee assetfinder.txt && cat *.txt | sort -u > all_subdomains.txt

2. Historical Endpoints

Discover historical endpoints and parameters
$ cat all_subdomains.txt | waybackurls | tee wayback.txt && cat all_subdomains.txt | gau --subs --threads 5 | tee gau.txt && cat wayback.txt gau.txt | sort -u | unfurl -u keys | tee -a params.txt

3. DNS & SSL Analysis

Deep DNS and SSL certificate analysis
$ dnsx -l all_subdomains.txt -json -a -aaaa -cname -mx -ptr -soa -txt -resp | tee dns_info.json && cat all_subdomains.txt | tlsx -cn -san -en | tee ssl_info.txt && cat all_subdomains.txt | dnsx -silent -d -resp-only | naabu -top-ports 1000 -silent | tee ports.txt

4. Enhanced Host Discovery

Advanced live host discovery and analysis
$ cat all_subdomains.txt | httpx -silent -title -tech-detect -status-code -follow-redirects -server -ip -cdn -asn -probe -csp-probe -tls-probe -favicon -ports 80,443,8080,8443 -json | tee httpx_results.json

5. Advanced Content Discovery

Multi-tool content discovery
$ cat httpx_results.json | jq -r '.url' | gospider -S - --js -t 50 --sitemap --robots -w -r | tee gospider.txt && ffuf -w /path/to/wordlist.txt -u FUZZ -mc 200,204,301,302,307,401,403,405 -recursion -recursion-depth 2 -c -v -t 50 | tee ffuf.txt

6. Parameter Analysis

Advanced parameter discovery and testing
$ cat httpx_results.json | jq -r '.url' | arjun -i - -oJ arjun.json && cat params.txt | qsreplace "FUZZ" | tee fuzz_ready.txt && for url in $(cat fuzz_ready.txt); do ffuf -u $url -w /path/to/payloads.txt -mr "FUZZ" -t 50; done

7. Advanced Vulnerability Scan

Comprehensive vulnerability scanning
$ cat httpx_results.json | jq -r '.url' | nuclei -t nuclei-templates -severity critical,high,medium -rl 50 -c 25 -mhe 10 -o nuclei.txt && jaeles scan -s /path/to/jaeles-signatures -U httpx_results.json -L 50 -v -o jaeles_results

8. JS Analysis

Deep JavaScript analysis
$ cat httpx_results.json | jq -r '.url' | getJS --complete --resolve | tee js_files.txt && cat js_files.txt | xargs -I% sh -c 'echo "Analyzing %"; trufflehog filesystem % --json | tee -a js_secrets.json'

9. Pattern Analysis

Advanced pattern matching and analysis
$ cat js_files.txt wayback.txt gau.txt gospider.txt | gf xss | tee gf_xss.txt && cat js_files.txt wayback.txt gau.txt gospider.txt | gf ssrf | tee gf_ssrf.txt && cat js_files.txt wayback.txt gau.txt gospider.txt | gf idor | tee gf_idor.txt && cat js_files.txt wayback.txt gau.txt gospider.txt | gf debug-pages | tee gf_debug.txt

10. Advanced Report

Generate comprehensive report
$ echo "# Bug Bounty Report for {domain}\n\n## Overview\n" > report.md && echo "### Scope\n- Target: {domain}\n- Start Date: $(date)\n\n## Findings\n" >> report.md && for f in *.txt *.json; do echo "### $f\n\`\`\`" >> report.md && cat $f >> report.md && echo "\`\`\`\n" >> report.md; done && echo "## Summary\n- Total Subdomains: $(cat all_subdomains.txt | wc -l)\n- Live Hosts: $(cat httpx_results.json | wc -l)\n- Vulnerabilities: $(cat nuclei.txt | wc -l)" >> report.md

Complete Bug Bounty Workflow

1. Subfinder Basic

Quick subdomain enumeration with Subfinder
$ subfinder -d {domain} -o subdomains.txt

2. Amass Deep Scan

Deep subdomain enumeration with Amass
$ amass enum -d {domain} -active -brute -w /path/to/wordlist.txt -o amass_results.txt

3. Combine & Sort Subdomains

Combine and sort all found subdomains
$ cat subdomains.txt amass_results.txt | sort -u > all_subdomains.txt

4. DNS Records

Get all DNS records
$ for sub in $(cat all_subdomains.txt); do dig +short $sub A AAAA CNAME MX NS SOA TXT; done > dns_records.txt

5. WHOIS & SSL Info

Gather WHOIS and SSL certificate info
$ whois {domain} > whois.txt && curl -s https://{domain} | openssl x509 -noout -text > ssl_info.txt

6. HTTPX Live Check

Check for live subdomains and gather info
$ cat all_subdomains.txt | httpx -title -tech-detect -status-code -follow-redirects -o httpx_results.txt

7. Dirsearch Scan

Directory and file enumeration
$ cat httpx_results.txt | grep 200 | cut -d' ' -f1 | dirsearch -l - -e php,asp,aspx,jsp,html,zip,jar -o dirsearch_results.txt

8. Nuclei Templates

Scan with Nuclei templates
$ cat httpx_results.txt | nuclei -t nuclei-templates -severity critical,high,medium -o nuclei_results.txt

9. GF Pattern Matching

Search for vulnerable patterns
$ cat dirsearch_results.txt | gf xss | tee gf_xss.txt && cat dirsearch_results.txt | gf ssrf | tee gf_ssrf.txt && cat dirsearch_results.txt | gf idor | tee gf_idor.txt

10. Katana Crawling

Smart web crawling
$ katana -list httpx_results.txt -jc -kf all -o katana_results.txt

11. Parameter Discovery

Discover hidden parameters
$ cat katana_results.txt | unfurl format %s://%d%p | urldedupe -qs | tee params.txt

12. CORS Check

Check for CORS misconfigurations
$ corsy -i httpx_results.txt -t 50 -o cors_results.txt

13. Quick Technology Scan

Identify web technologies
$ whatweb -i httpx_results.txt -a 3 --no-errors | tee whatweb_results.txt

14. URL Manipulation

Replace query string values
$ cat params.txt | qsreplace "FUZZ" | tee fuzz_ready_urls.txt

15. Generate Report

Combine all results into a report
$ echo "Bug Bounty Report for {domain}" > report.md && echo "## Findings" >> report.md && for f in *_results.txt; do echo "### $f" >> report.md && cat $f >> report.md && echo "\n---\n" >> report.md; done
